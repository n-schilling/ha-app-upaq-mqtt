abi <abi/3.0>,
#include <tunables/global>

# Talks to the UniFi console and the MQTT broker; writes only its sensor list
# and the pinned certificate to /data. s6-overlay runs as root and hands the
# data folder to user upaq, as which the bridge runs. Tested on a real installation.
profile upaq_mqtt flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # s6-overlay, and the hand-over of /data to the bridge's user
  capability chown,
  capability dac_override,
  capability fowner,
  capability kill,
  capability setgid,
  capability setuid,

  network inet dgram,
  network inet6 dgram,
  network inet stream,
  network inet6 stream,
  network unix,

  # s6 stops its services; the Supervisor stops the container
  signal (send, receive),

  # s6-overlay
  /init rix,
  /package/** rix,
  /command/** rix,
  /etc/s6-overlay/** rix,
  /run/** rwk,
  # s6-overlay runs the scripts it copied to /run at the start
  /run/s6/** rix,
  /run/s6-rc*/** rix,
  /run/service/** rix,
  /run/ rw,

  # Programs and libraries of the image
  /bin/** rix,
  /sbin/** rix,
  /usr/bin/** rix,
  /usr/sbin/** rix,
  /usr/local/bin/** rix,
  /usr/lib/ r,
  /usr/lib/** rm,
  /usr/lib/bashio/bashio rix,
  /usr/local/lib/** rm,
  /lib/** rm,
  /usr/share/** r,
  /etc/** r,

  # bashio
  /tmp/ r,
  /tmp/** rwk,

  # Options, sensor list, pinned certificate
  /data/ rw,
  /data/** rwk,
  # mqtt_ca, read only
  /ssl/ r,
  /ssl/** r,

  @{PROC}/@{pid}/** r,
  @{PROC}/ r,
  /dev/null rw,
  /dev/urandom r,
  /dev/random r,
  /dev/tty rw,

  deny /proc/sys/** w,
  deny /sys/** w,
}
